```
with next_task as (
    select 
    substr(
        body, 
        instr(body, "4. ") + length("4. "), 
        instr(substr(body, instr(body, "4. ")), "TODO")
    ) as title,

    (substr(
        body, 
        instr(body, "4. ") + length("4. "), 
        instr(substr(body, instr(body, "4. ")), "TODO") -4) || "DONE"
    ) as marked_done
    from workplan where workflow_id = 10
    limit 1
)
select 
REPLACE(
    body, 
    (select title from next_task limit 1),
    (select marked_done from next_task limit 1)
)
from workplan
WHERE workflow_id = 10;
```
